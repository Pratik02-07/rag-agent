    #!/bin/bash

    # ============================================================
    # RAG Agent - EC2 Deployment Script
    # ============================================================
    # Tasks:
    #   1. Install Git if missing
    #   2. Install Docker if missing
    #   3. Install Nginx if missing
    #   4. Clone/update RAG Agent repository
    #   5. Build Docker images
    #   6. Start containers with Docker Compose
    #   7. Verify deployment
    #   8. Handle errors safely
    # ============================================================

    set -e

    REPO_URL="https://github.com/Pratik02-07/rag-agent.git"
    APP_DIR="$HOME/rag-agent"

    echo "============================================================"
    echo "        RAG AGENT DEPLOYMENT STARTED"
    echo "============================================================"

    # ------------------------------------------------------------
    # Helper functions
    # ------------------------------------------------------------

    log() {
        echo
        echo "[INFO] $1"
    }

    success() {
        echo "[SUCCESS] $1"
    }

    error() {
        echo "[ERROR] $1"
        exit 1
    }

    command_exists() {
        command -v "$1" >/dev/null 2>&1
    }

    # ------------------------------------------------------------
    # Detect OS
    # ------------------------------------------------------------

    if [ -f /etc/os-release ]; then
        . /etc/os-release
        OS="$ID"
    else
        error "Unable to detect operating system."
    fi

    log "Detected operating system: $OS"

    # ------------------------------------------------------------
    # Install Git
    # ------------------------------------------------------------

    install_git() {

        if command_exists git; then
            success "Git is already installed."
            return
        fi

        log "Installing Git..."

        case "$OS" in

            ubuntu|debian)
                sudo apt-get update
                sudo apt-get install -y git
                ;;

            amzn)
                sudo dnf install -y git
                ;;

            centos|rhel|fedora)
                sudo dnf install -y git
                ;;

            *)
                error "Unsupported OS for Git installation: $OS"
                ;;
        esac

        command_exists git || error "Git installation failed."

        success "Git installed successfully."
    }

    # ------------------------------------------------------------
    # Install Docker
    # ------------------------------------------------------------

    install_docker() {

        if command_exists docker; then
            success "Docker is already installed."
        else

            log "Installing Docker..."

            case "$OS" in

                ubuntu|debian)
                    sudo apt-get update
                    sudo apt-get install -y docker.io
                    ;;

                amzn)
                    sudo dnf install -y docker
                    ;;

                centos|rhel|fedora)
                    sudo dnf install -y docker
                    ;;

                *)
                    error "Unsupported OS for Docker installation: $OS"
                    ;;
            esac
        fi

        log "Starting Docker service..."

        sudo systemctl enable docker
        sudo systemctl start docker

        if ! sudo systemctl is-active --quiet docker; then
            error "Docker service failed to start."
        fi

        success "Docker is running."

        # Allow current user to run Docker without sudo
        sudo usermod -aG docker "$USER" || true
    }

    # ------------------------------------------------------------
    # Install Docker Compose
    # ------------------------------------------------------------

    install_docker_compose() {

        if docker compose version >/dev/null 2>&1; then
            success "Docker Compose plugin is already installed."
            return
        fi

        log "Installing Docker Compose..."

        case "$OS" in

            ubuntu|debian)
                sudo apt-get update
                sudo apt-get install -y docker-compose-plugin
                ;;

            amzn)
                # Docker Compose plugin package may not be available
                # on every Amazon Linux version.
                if ! docker compose version >/dev/null 2>&1; then
                    log "Installing Docker Compose plugin manually..."

                    sudo mkdir -p /usr/local/lib/docker/cli-plugins

                    sudo curl -SL \
                        https://github.com/docker/compose/releases/latest/download/docker-compose-linux-x86_64 \
                        -o /usr/local/lib/docker/cli-plugins/docker-compose

                    sudo chmod +x \
                        /usr/local/lib/docker/cli-plugins/docker-compose
                fi
                ;;

            *)
                error "Unable to install Docker Compose automatically on: $OS"
                ;;
        esac

        docker compose version >/dev/null 2>&1 \
            || error "Docker Compose installation failed."

        success "Docker Compose installed."
    }

    # ------------------------------------------------------------
    # Install Nginx
    # ------------------------------------------------------------

    install_nginx() {

        if command_exists nginx; then
            success "Nginx is already installed."
            return
        fi

        log "Installing Nginx..."

        case "$OS" in

            ubuntu|debian)
                sudo apt-get update
                sudo apt-get install -y nginx
                ;;

            amzn)
                sudo dnf install -y nginx
                ;;

            centos|rhel|fedora)
                sudo dnf install -y nginx
                ;;

            *)
                error "Unsupported OS for Nginx installation: $OS"
                ;;
        esac

        sudo systemctl enable nginx

        success "Nginx installed."
    }

    # ------------------------------------------------------------
    # Clone / Update Repository
    # ------------------------------------------------------------

    clone_repository() {

        if [ -d "$APP_DIR/.git" ]; then

            log "Repository already exists."

            cd "$APP_DIR"

            log "Updating repository..."

            git fetch origin
            git pull origin main

        else

            log "Cloning RAG Agent repository..."

            git clone "$REPO_URL" "$APP_DIR"

            cd "$APP_DIR"
        fi

        success "Repository ready at $APP_DIR."
    }

    # ------------------------------------------------------------
    # Check required project files
    # ------------------------------------------------------------

    check_project() {

        cd "$APP_DIR"

        log "Checking project files..."

        if [ ! -f "docker-compose.yml" ] && [ ! -f "compose.yml" ]; then
            error "docker-compose.yml / compose.yml not found."
        fi

        if [ ! -d "be-rag" ]; then
            error "Backend directory 'be-rag' not found."
        fi

        if [ ! -d "Fe-rag" ]; then
            error "Frontend directory 'Fe-rag' not found."
        fi

        success "Project structure looks valid."
    }

    # ------------------------------------------------------------
    # Build Docker images
    # ------------------------------------------------------------

    build_images() {

        cd "$APP_DIR"

        log "Building Docker images..."

        docker compose build --no-cache

        success "Docker images built successfully."
    }

    # ------------------------------------------------------------
    # Start application
    # ------------------------------------------------------------

    start_application() {

        cd "$APP_DIR"

        log "Starting RAG Agent containers..."

        docker compose up -d

        sleep 10

        success "Docker Compose started."
    }

    # ------------------------------------------------------------
    # Check containers
    # ------------------------------------------------------------

    check_containers() {

        cd "$APP_DIR"

        log "Checking running containers..."

        docker compose ps

        RUNNING=$(docker compose ps --status running --quiet | wc -l)

        if [ "$RUNNING" -lt 3 ]; then

            echo
            echo "Container logs:"
            docker compose logs --tail=100

            error "Not all RAG Agent containers are running."

        fi

        success "All RAG Agent containers are running."
    }

    # ------------------------------------------------------------
    # Check Docker services
    # ------------------------------------------------------------

    check_services() {

        log "Checking Docker..."

        docker --version

        log "Checking Docker Compose..."

        docker compose version

        log "Checking Nginx..."

        nginx -v

        success "Required services are installed."
    }

    # ------------------------------------------------------------
    # Main deployment
    # ------------------------------------------------------------

    install_git

    install_docker

    install_docker_compose

    install_nginx

    check_services

    clone_repository

    check_project

    build_images

    start_application

    check_containers

    # ------------------------------------------------------------
    # Final information
    # ------------------------------------------------------------

    echo
    echo "============================================================"
    echo "        RAG AGENT DEPLOYMENT COMPLETED"
    echo "============================================================"

    echo
    echo "Application directory:"
    echo "$APP_DIR"

    echo
    echo "Running containers:"
    docker compose -f "$APP_DIR/docker-compose.yml" ps 2>/dev/null || \
    docker compose -f "$APP_DIR/compose.yml" ps

    echo
    echo "Frontend:"
    echo "http://$(hostname -I | awk '{print $1}'):3000"

    echo
    echo "Backend:"
    echo "http://$(hostname -I | awk '{print $1}'):5000"

    echo
    echo "Ollama:"
    echo "http://$(hostname -I | awk '{print $1}'):11434"

    echo
    echo "============================================================"
    echo "IMPORTANT:"
    echo "If this is the first deployment, log out and log back in"
    echo "after adding your user to the docker group."
    echo "============================================================"
